create table "aelina"."Users"(
    id uuid primary key ,
    login varchar(255) not null unique ,
    email varchar(255) not null unique ,
    password_hash text not null,
    created_at timestamp not null
);

DROP TABLE aelina."Folders" CASCADE ;


create table "aelina"."Folders"(
    id uuid primary key ,
    title varchar(255) not null ,
    owner_id uuid not null,
    parent_folder_id uuid null,
    created_at timestamp not null,

    constraint fk_folders_owner foreign key (owner_id) references aelina."Users"(id) on delete cascade,
    constraint fk_folders_parents foreign key (parent_folder_id) references aelina."Folders"(id) on delete cascade
);

create table "aelina"."Notes"(
    id uuid primary key ,
    title varchar(255) not null,
    content text not null,
    owner_id uuid not null,
    folder_id uuid null,

    created_at timestamp not null,
    updated_at timestamp not null,

    is_favorite boolean not null  DEFAULT FALSE,
    is_archive boolean not null default false,

    timer timestamp null,
    note_type varchar(50) null,

    constraint fk_notes_owner foreign key (owner_id) references aelina."Users"(id) on delete cascade,
    constraint fk_notes_folder foreign key (folder_id) references  aelina."Folders"(id) on delete set null
);

CREATE OR REPLACE PROCEDURE aelina.update_note(
    n_id UUID,
    n_title VARCHAR,
    n_content VARCHAR,
    n_folder_id UUID DEFAULT NULL,
    n_updated_at TIMESTAMPTZ DEFAULT NULL,
    n_is_favorite BOOLEAN DEFAULT NULL,
    n_is_archive BOOLEAN DEFAULT NULL,
    n_note_type VARCHAR DEFAULT NULL
)
LANGUAGE plpgsql
    AS $$
    BEGIN
        UPDATE aelina."Notes"
        SET
            title = n_title,
            content = n_content,
            folder_id = COALESCE(n_folder_id, folder_id),
            updated_at = COALESCE(n_updated_at, now()),
            is_favorite = COALESCE(n_is_favorite, is_favorite),
            is_archive = COALESCE(n_is_archive, is_archive),
            note_type = COALESCE(n_note_type, note_type)
        WHERE id = n_id;
    END;
    $$;

CALL aelina.update_note('d12c5cf3-13ba-4d13-ba78-48bcc659d471', 'Заметка тест', 'text');
